\section{Sơ đồ Tuần tự (Sequence Diagram)}

Mục này mô tả vòng đời của (1) phiên kết nối, (2) một tin nhắn chat và (3) một lần truyền file. Các sơ đồ dùng ba đối tượng: \textbf{Client A} (người gửi), \textbf{Server} và \textbf{Client B} (người nhận).

% ---- helpers cho sơ đồ tuần tự ----
\newcommand{\seqlife}[3]{%  x, tên, chiều cao
  \node[draw, fill=gray!15, minimum width=2.4cm, font=\small\bfseries] at (#1,0) {#2};
  \draw[dashed, gray] (#1,-0.35) -- (#1,-#3);}
\newcommand{\seqmsg}[5]{%  x1, x2, y, nhãn, kiểu
  \draw[#5, ->, >=stealth] (#1,-#3) -- node[above, font=\scriptsize, align=center]{#4} (#2,-#3);}
\newcommand{\seqnote}[4]{%  x, y, rộng, nội dung
  \node[draw, fill=yellow!20, font=\scriptsize, text width=#3, align=center, inner sep=2pt] at (#1,-#2) {#4};}
